cmake_minimum_required(VERSION 3.20)
project(AuroraPrototype VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(aurora-core src/cpu.cpp src/media.cpp src/ps1_memory.cpp src/disc.cpp src/execution.cpp src/homebrew.cpp)
target_include_directories(aurora-core PUBLIC include)
target_compile_definitions(aurora-core PUBLIC AURORA_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(aurora-core PRIVATE /W4 /permissive-)
else()
  target_compile_options(aurora-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(aurora-cli src/cli.cpp)
target_link_libraries(aurora-cli PRIVATE aurora-core)

if(WIN32)
  find_path(VULKAN_HEADERS_DIR vulkan/vulkan.h REQUIRED)
  add_executable(Aurora WIN32 src/windows_app.cpp src/vulkan_presenter.cpp)
  target_include_directories(Aurora PRIVATE "${VULKAN_HEADERS_DIR}")
  target_compile_definitions(Aurora PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(Aurora PRIVATE aurora-core comctl32 comdlg32 shell32 ole32 uuid gdi32 user32)
  if(MINGW)
    target_link_options(Aurora PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(aurora-cli PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  foreach(suite cpu media system disc io homebrew)
    add_executable(${suite}-tests tests/${suite}_tests.cpp)
    target_link_libraries(${suite}-tests PRIVATE aurora-core)
    add_test(NAME ${suite} COMMAND ${suite}-tests)
  endforeach()
endif()
